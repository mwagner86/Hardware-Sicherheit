% Wiederverwendbares gruppiertes Balkendiagramm fuer eine Metrik.
% #1=Titel #2=ylabel #3=Spalte Baseline #4=Spalte NoisyNeighbor #5=ymax
\newcommand{\metricplot}[5]{%
\begin{axis}[
    title={#1},
    ybar, bar width=9pt,
    width=7.5cm, height=5cm,
    enlarge x limits=0.30,
    ylabel={#2}, ymin=0, ymax=#5,
    symbolic x coords={QEMU, KVM, LXC},
    xtick=data,
    nodes near coords, nodes near coords align={vertical},
    every node near coord/.append style={font=\tiny, /pgf/number format/fixed},
    x tick label style={font=\small},
    title style={font=\small\bfseries},
    legend style={at={(0.5,1.30)}, anchor=north, legend columns=-1, font=\footnotesize},
]
\addplot table [x=Virtualisierung, y=#3] {\data};
\addplot table [x=Virtualisierung, y=#4] {\data};
\legend{Baseline, Noisy Neighbor}
\end{axis}}
